class KCoinPickup : CustomInventory
{
	Default
	{
		Radius 12; Height 16; +INVENTORY.ALWAYSPICKUP;
		Inventory.PickupSound "kc/coin"; Inventory.PickupMessage "";
		Scale 0.9;
	}
	States
	{
	Spawn:
		COIN ABCB 5 Bright;
		Loop;
	Pickup:
		TNT1 A 0
		{
			A_GiveInventory("KCoin", 1);
			A_GiveInventory("Health", 3);
			for (int i = 0; i < 5; i++)
				A_SpawnParticle(0xFFD23A, SPF_FULLBRIGHT, 18, 4, 0, frandom(-6,6), frandom(-6,6), frandom(10,24), frandom(-1,1), frandom(-1,1), frandom(1,3), 0, 0, -0.2, 1, -0.05);
		}
		Stop;
	}
	override void PostBeginPlay()
	{
		Super.PostBeginPlay();
		// coins pop up out of whatever dropped them, then settle
		vel = (frandom(-3, 3), frandom(-3, 3), frandom(4, 7));
	}
}

class Greedling : Actor replaces DoomImp
{
	Default
	{
		Health 60; Radius 20; Height 52; Speed 5; PainChance 160; Mass 100; Scale 1.1;
		MONSTER; +FLOORCLIP;
		SeeSound "kc/greedsight"; PainSound "kc/greedsight"; DeathSound "kc/greeddie"; ActiveSound "kc/greedsight";
		Tag "Greed"; Obituary "%o was robbed by the Greed."; MeleeRange 56;
	}

	void InkBurst(int n)
	{
		for (int i = 0; i < n; i++)
			A_SpawnParticle(0x7A3FA0, 0, 22, 5, 0, frandom(-8,8), frandom(-8,8), frandom(8,40), frandom(-3,3), frandom(-3,3), frandom(1,4), 0, 0, -0.25, 1, -0.04);
	}

	void DropCoins(int n)
	{
		for (int i = 0; i < n; i++) Spawn("KCoinPickup", pos + (0, 0, 16));
	}

	// The Greed snatches a coin out of the player's purse: it pops out on the floor.
	void Snatch()
	{
		if (!target || target.player == null) return;
		if (Distance3D(target) > MeleeRange + target.radius + 8) return;
		if (target.CountInv("KCoin") > 0)
		{
			target.TakeInventory("KCoin", 1);
			let c = Spawn("KCoinPickup", target.pos + (0, 0, 40));
			if (c) c.vel = (frandom(-4,4), frandom(-4,4), 8);
		}
	}

	States
	{
	Spawn:
		GRDL A 10 A_Look;
		Loop;
	See:
		GRDL A 4 A_Chase;
		GRDL B 4 A_Chase;
		GRDL C 4 A_Chase;
		GRDL D 4 A_Chase;
		Loop;
	Melee:
		GRDL E 4 A_FaceTarget;
		GRDL F 6 { A_CustomMeleeAttack(random(2, 4) * 3, "kc/greedbite", "kc/greedbite"); Snatch(); }
		Goto See;
	Missile:
		GRDL E 6 A_FaceTarget;
		GRDL F 3 { A_StartSound("kc/greedbite"); A_FaceTarget(); vel.z = 5; A_Recoil(-9); }
		GRDL F 10;
		Goto See;
	Pain:
		GRDL G 3 InkBurst(5);
		GRDL G 3 A_Pain;
		Goto See;
	Death:
		GRDL G 3 { A_Scream(); InkBurst(14); }
		GRDL H 6 { A_NoBlocking(); DropCoins(random(1, 2)); }
		GRDL H -1;
		Stop;
	}
}

class KingdomHandler : EventHandler
{
	int greedKills;
	int archers, vagrants;
	int toastUntil;
	String toastText;

	override void PlayerEntered(PlayerEvent e)
	{
		let p = players[e.PlayerNumber].mo;
		if (!p) return;
		p.GiveInventory("KBow", 1);
		p.A_SelectWeapon("KBow");
		p.GiveInventory("KCoin", 3);
		Toast("THE GREED IS COMING. PROTECT THE CROWN!", 120);
	}

	void Toast(String s, int tics)
	{
		toastText = s;
		toastUntil = level.maptime + tics;
	}

	override void WorldThingDied(WorldEvent e)
	{
		if (!e.Thing) return;
		if (e.Thing is "Greedling")
		{
			greedKills++;
			if (greedKills % 3 == 0) VagrantArrives();
		}
	}

	void VagrantArrives()
	{
		let p = players[consoleplayer].mo;
		if (!p || archers + vagrants >= 6) return;
		for (int i = 0; i < 8; i++)
		{
			double a = frandom(0, 360);
			double d = frandom(160, 260);
			let v = Actor.Spawn("KVagrant", p.pos + (cos(a) * d, sin(a) * d, 0));
			if (!v) continue;
			if (v.TestMobjLocation()) { Toast("A vagrant wanders in. Give them a coin!", 90); return; }
			v.Destroy();
		}
	}

	override void WorldTick()
	{
		let p = players[consoleplayer].mo;
		if (!p || level.maptime % 5) return;
		int a = 0, v = 0;
		let it = ThinkerIterator.Create("Actor");
		Actor m;
		while (m = Actor(it.Next()))
		{
			if (m is "KArcher" && m.health > 0) a++;
			else if (m is "KVagrant" && m.health > 0)
			{
				v++;
				if (p.CountInv("KCoin") > 0 && m.Distance3D(p) < 130) Recruit(m, p);
			}
		}
		archers = a; vagrants = v;
	}

	void Recruit(Actor vag, Actor p)
	{
		p.TakeInventory("KCoin", 1);
		let ar = Actor.Spawn("KArcher", vag.pos);
		if (ar)
		{
			ar.angle = vag.angle;
			ar.A_StartSound("kc/recruit", CHAN_VOICE);
			for (int i = 0; i < 16; i++)
				ar.A_SpawnParticle(0xFFD23A, SPF_FULLBRIGHT, 30, 6, 0, frandom(-8,8), frandom(-8,8), frandom(0,40), frandom(-2,2), frandom(-2,2), frandom(2,5), 0, 0, -0.1, 1, -0.03);
		}
		vag.Destroy();
		Toast("A vagrant became an Archer!", 90);
	}

	override void RenderOverlay(RenderEvent e)
	{
		let p = players[consoleplayer].mo;
		if (!p) return;
		screen.DrawTexture(TexMan.CheckForTexture("CRWNA0", TexMan.Type_Any), false, 6, 4, DTA_VirtualWidth, 320, DTA_VirtualHeight, 200, DTA_TopOffset, 0, DTA_LeftOffset, 0, DTA_DestWidth, 30, DTA_DestHeight, 24);
		screen.DrawTexture(TexMan.CheckForTexture("COINA0", TexMan.Type_Any), false, 42, 8, DTA_VirtualWidth, 320, DTA_VirtualHeight, 200, DTA_TopOffset, 0, DTA_LeftOffset, 0, DTA_DestWidth, 16, DTA_DestHeight, 16);
		screen.DrawText(SmallFont, Font.CR_GOLD, 31, 3, String.Format("x %d", p.CountInv("KCoin")), DTA_VirtualWidth, 160, DTA_VirtualHeight, 100);
		screen.DrawText(SmallFont, Font.CR_TAN, 8, 32, String.Format("Archers: %d", archers), DTA_VirtualWidth, 320, DTA_VirtualHeight, 200);
		if (level.maptime < toastUntil)
			screen.DrawText(BigFont, Font.CR_GOLD, 160 - BigFont.StringWidth(toastText) / 2, 40, toastText, DTA_VirtualWidth, 320, DTA_VirtualHeight, 200);
	}
}
